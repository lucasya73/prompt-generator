<script>
function val(id) {
 return document.getElementById(id).value.trim();
}

function lines(text) {
 return text ? text.split(/\r?\n/).map(s => s.trim()).filter(Boolean) : [];
}

function bullet(text, prefix = '- ') {
 const arr = lines(text);
 return arr.length ? arr.map(s => prefix + s).join('\n') : '';
}

function fallback(text, fb) {
 return text || fb;
}

function generate() {
 const model = val('model');
 const role = fallback(val('role'), '专业助手');
 const task = fallback(val('task'), '请完成用户指定的任务。');
 const context = val('context');
 const input = val('input');
 const constraints = val('constraints');
 const outputFormat = fallback(val('outputFormat'), '使用清晰、结构化的 Markdown 输出。');
 const examples = val('examples');
 const tone = val('tone');
 const step = val('step');

 const constraintBlock = bullet(constraints) || '- 遵循用户要求，不要编造。';
 const toneLine = tone ? `\n- 语气风格：${tone}` : '';

 const stepText =
 step === 'internal'
 ? '请在内部逐步推理，但最终只输出结果，不要展示推理过程。'
 : step === 'yes'
 ? '请展示关键推理步骤，并给出最终结果。'
 : '请直接给出最终结果。';

 let prompt = '';

 if (model === 'gpt') {
 prompt = `# 角色
你是一位${role}。

# 任务
${task}

${context ? `# 背景\n${context}\n\n` : ''}${input ? `# 输入\n${input}\n\n` : ''}# 约束
${constraintBlock}${toneLine}

# 输出格式
${outputFormat}

${examples ? `# 示例\n${examples}\n\n` : ''}# 执行要求
${stepText}
最终输出必须严格满足上述格式与约束。`;
 } else if (model === 'claude') {
 prompt = `<role>
你是一位${role}。
</role>

${context ? `<context>\n${context}\n</context>\n\n` : ''}${input ? `<input>\n${input}\n</input>\n\n` : ''}<task>
${task}
</task>

<constraints>
${constraintBlock}${toneLine}
</constraints>

<output_format>
${outputFormat}
</output_format>

${examples ? `<examples>\n${examples}\n</examples>\n\n` : ''}<thinking_instruction>
${stepText}
</thinking_instruction>

请严格按照上述 XML 标签中的要求执行。`;
 } else if (model === 'gemini') {
 prompt = `## 角色
你是一位${role}。

## 目标
${task}

${context ? `## 背景\n${context}\n\n` : ''}${input ? `## 输入\n${input}\n\n` : ''}## 约束
${constraintBlock}${toneLine}

## 输出格式
${outputFormat}

${examples ? `## 示例\n${examples}\n\n` : ''}## 执行步骤
1. 理解目标、背景和输入。
2. 按约束条件处理内容。
3. 生成符合输出格式的最终答案。
${stepText}

请现在开始。`;
 } else {
 prompt = `# 角色
你是一位${role}。

# 任务
${task}

${context ? `# 背景\n${context}\n\n` : ''}${input ? `# 输入\n${input}\n\n` : ''}# 约束
${constraintBlock}${toneLine}

# 输出格式
${outputFormat}

${examples ? `# 示例\n${examples}\n\n` : ''}# 执行要求
${stepText}
请严格按以上结构执行。`;
 }

 document.getElementById('result').value = prompt;
}

function copyPrompt() {
 const result = document.getElementById('result');
 if (!result.value.trim()) {
 alert('请先生成提示词');
 return;
 }

 result.select();
 result.setSelectionRange(0, result.value.length);

 if (navigator.clipboard && navigator.clipboard.writeText) {
 navigator.clipboard.writeText(result.value)
 .then(() => alert('已复制到剪贴板'))
 .catch(() => {
 document.execCommand('copy');
 alert('已复制到剪贴板');
 });
 } else {
 document.execCommand('copy');
 alert('已复制到剪贴板');
 }
}
